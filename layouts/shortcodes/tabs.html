{{- /* Tabbed content. Tabs whose label names an OS (Linux, macOS, Windows) open automatically for readers on that OS, and choosing a tab selects the tab with the same label everywhere on the page. @param items comma-separated tab labels (required) @param defaultIndex tab shown first (default 0) @example {{< tabs items="Linux,macOS,Windows" >}}{{< tab >}}...{{< /tab >}}{{< /tabs >}} */ -}} {{- if not (.Get "items") }}{{ errorf "tabs shortcode: 'items' is required (%s)" .Position }}{{ end -}} {{- $items := split (.Get "items") "," -}} {{- $default := int (.Get "defaultIndex" | default "0") -}}
{{- range $i, $item := $items }} {{- $label := trim $item " " }} {{- $l := lower $label }} {{- $os := slice }} {{- if in $l "linux" }}{{ $os = $os | append "linux" }}{{ end }} {{- if in $l "macos" }}{{ $os = $os | append "mac" }}{{ end }} {{- if in $l "windows" }}{{ $os = $os | append "windows" }}{{ end }} {{- end }}
{{- .Inner -}}
