{{- /* Renders a CSV file as a table. @param 0 path of the CSV file, relative to the project root @param 1 delimiter (default ",") @param 2 "true" when the first row is a header (default "true") Status symbols in cells are shown as words (Yes / No / Limited). */ -}} {{- $csvFile := .Get 0 -}} {{- $delimiter := .Get 1 | default "," -}} {{- $hasHeader := .Get 2 | default "true" -}} {{- $csvContent := readFile $csvFile | transform.Unmarshal (dict "delimiter" $delimiter) -}} {{- if not $csvContent }} {{- errorf "csv-table: %q is empty or missing (%s)" $csvFile .Position -}} {{- else }}
{{- if eq $hasHeader "true" }} {{ range index $csvContent 0 }}{{ end }} {{- range after 1 $csvContent }} {{ range . }}{{ partial "csvcell.html" . }}{{ end }} {{- end }} {{- else }} {{- range $csvContent }} {{ range . }}{{ partial "csvcell.html" . }}{{ end }} {{- end }} {{- end }}
{{ trim . " " }}
{{- end }}
